cmake_minimum_required(VERSION 3.16)
project(alive_lio LANGUAGES C CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(Sophus REQUIRED)
find_package(OpenMP REQUIRED)
find_package(livox_ros_driver2 QUIET)

set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C/C++ distribution (CPU or GPU)")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
  HINTS "${ONNXRUNTIME_ROOT}/include" "$ENV{ONNXRUNTIME_ROOT}/include")
find_library(ONNXRUNTIME_LIBRARY onnxruntime
  HINTS "${ONNXRUNTIME_ROOT}/lib" "$ENV{ONNXRUNTIME_ROOT}/lib")
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
  message(FATAL_ERROR "ONNX Runtime not found. Set -DONNXRUNTIME_ROOT=/path/to/onnxruntime-linux-x64-VERSION")
endif()
get_filename_component(ONNXRUNTIME_LIBRARY_DIR "${ONNXRUNTIME_LIBRARY}" DIRECTORY)
set(ros_dependencies rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs
  visualization_msgs builtin_interfaces tf2_ros pcl_conversions ament_index_cpp)

add_executable(alive_lio_node src/voxelMapping.cpp src/preprocess.cpp)
target_include_directories(alive_lio_node PRIVATE include ${PCL_INCLUDE_DIRS} ${ONNXRUNTIME_INCLUDE_DIR})
ament_target_dependencies(alive_lio_node ${ros_dependencies})
target_link_libraries(alive_lio_node Eigen3::Eigen Sophus::Sophus ${PCL_LIBRARIES}
  OpenMP::OpenMP_CXX "${ONNXRUNTIME_LIBRARY}")
# Keep deployed executables independent of an LD_LIBRARY_PATH override.
set_target_properties(alive_lio_node PROPERTIES INSTALL_RPATH "${ONNXRUNTIME_LIBRARY_DIR}")
if(livox_ros_driver2_FOUND)
  ament_target_dependencies(alive_lio_node livox_ros_driver2)
  target_compile_definitions(alive_lio_node PRIVATE ALIVE_WITH_LIVOX)
  message(STATUS "ALIVE-LIO: Livox CustomMsg enabled")
else()
  message(STATUS "ALIVE-LIO: building PointCloud2 support without Livox CustomMsg")
endif()
# Bound parallel work within each scan; DDS and inference use additional threads.
target_compile_definitions(alive_lio_node PRIVATE MP_EN MP_PROC_NUM=4)

foreach(converter pandar_to_velodyne rs_to_velodyne)
  add_executable(${converter} src/converter/${converter}.cpp)
  target_include_directories(${converter} PRIVATE include ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(${converter} rclcpp sensor_msgs pcl_conversions builtin_interfaces)
  target_link_libraries(${converter} ${PCL_LIBRARIES})
  install(TARGETS ${converter} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(TARGETS alive_lio_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY onnx/model DESTINATION share/${PROJECT_NAME}/onnx)
install(FILES README.md README_ROS2.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
